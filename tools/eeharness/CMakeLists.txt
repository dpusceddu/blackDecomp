cmake_minimum_required(VERSION 3.20)
project(black_eeharness LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(EEHARNESS_SANITIZERS "Use address and undefined-behavior sanitizers" OFF)
find_package(ZLIB REQUIRED)
add_library(ee_core decode.cpp memory.cpp cpu.cpp cop0.cpp fpu.cpp vu.cpp replay.cpp event_journal.cpp differential.cpp)
target_link_libraries(ee_core PUBLIC ZLIB::ZLIB)
target_include_directories(ee_core PUBLIC .)
target_compile_options(ee_core PRIVATE -Wall -Wextra -Wpedantic -Werror -Wno-missing-field-initializers)
if(EEHARNESS_SANITIZERS)
    target_compile_options(ee_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(ee_core PUBLIC -fsanitize=address,undefined)
endif()
add_executable(eeharness main.cpp vu_probe.cpp fixture.cpp)
target_link_libraries(eeharness PRIVATE ee_core)
add_executable(ee_semantics tests.cpp)
target_link_libraries(ee_semantics PRIVATE ee_core)
add_executable(vu_semantics vu_tests.cpp)
target_link_libraries(vu_semantics PRIVATE ee_core)
add_executable(cop0_semantics cop0_tests.cpp)
target_link_libraries(cop0_semantics PRIVATE ee_core)
enable_testing()
add_test(NAME ee_semantics COMMAND ee_semantics)
add_test(NAME vu_semantics COMMAND vu_semantics)
add_test(NAME cop0_semantics COMMAND cop0_semantics)
add_executable(replay_semantics replay_tests.cpp)
target_link_libraries(replay_semantics PRIVATE ee_core)
add_test(NAME replay_semantics COMMAND replay_semantics)
add_executable(differential_semantics differential_tests.cpp)
target_link_libraries(differential_semantics PRIVATE ee_core)
add_test(NAME differential_semantics COMMAND differential_semantics)
